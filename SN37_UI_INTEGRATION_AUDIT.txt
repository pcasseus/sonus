class HomePage extends StatelessWidget {
  const HomePage({super.key});

  @override
  Widget build(BuildContext context) {
    final controller = context.watch<HomeController?>();

    if (controller == null) {
      return const _HomeMessage(
        icon: Icons.hourglass_empty_rounded,
        title: 'Home is getting ready',
        message: 'Your Sonus Home data is still initializing.',
      );
    }

    final snapshot = controller.snapshot;

    if (controller.isLoading && snapshot == null) {
      return const Center(child: CircularProgressIndicator());
    }

    if (controller.hasError && snapshot == null) {
      return _HomeError(onRetry: controller.load);
    }

    if (snapshot == null || snapshot.isEmpty) {
      return RefreshIndicator(
        onRefresh: controller.refresh,
        child: ListView(
          physics: const AlwaysScrollableScrollPhysics(),
          padding: const EdgeInsets.fromLTRB(20, 28, 20, 40),
          children: const [
            SizedBox(height: 100),
            _HomeMessage(
              icon: Icons.home_outlined,
              title: 'Your Home is empty',
              message: 'Import music to get started.',
            ),
          ],
        ),
      );
    }

    return RefreshIndicator(
      onRefresh: controller.refresh,
      child: ListView(
        key: const PageStorageKey<String>('home-scroll'),
        physics: const AlwaysScrollableScrollPhysics(),
        padding: const EdgeInsets.fromLTRB(0, 20, 0, 40),
        children: [
          if (controller.hasError)
            Padding(
              padding: const EdgeInsets.fromLTRB(20, 0, 20, 12),
              child: _RefreshErrorBanner(onRetry: controller.refresh),
            ),

          if (snapshot.weeklyPopularTracks.isNotEmpty) ...[
            const _SectionHeading(title: 'Popular This Week'),
            const SizedBox(height: 12),
            _PopularTracksRail(entries: snapshot.weeklyPopularTracks),
            const SizedBox(height: 30),
          ],

          if (snapshot.weeklyTopArtists.isNotEmpty) ...[
            const _SectionHeading(title: 'Top Artists'),
            const SizedBox(height: 12),
            _TopArtistsRail(entries: snapshot.weeklyTopArtists),
            const SizedBox(height: 30),
          ],

          if (snapshot.recentlyAddedTracks.isNotEmpty) ...[
            const _SectionHeading(title: 'Recently Added'),
            const SizedBox(height: 12),
            _RecentlyAddedTracksRail(tracks: snapshot.recentlyAddedTracks),
            const SizedBox(height: 30),
          ],

          if (snapshot.recentlyAddedAlbums.isNotEmpty) ...[
            const _SectionHeading(title: 'New Albums'),
            const SizedBox(height: 12),
            _AlbumsRail(albums: snapshot.recentlyAddedAlbums),
          ],
        ],
      ),
    );
  }
}

Future<void> _showTrackContext(BuildContext context, LibraryTrack track) async {
  if (track.isBlocked || !track.isAvailable) {
    return;
  }

  final playback = context.read<PlaybackCoordinator?>();

  await showModalBottomSheet<void>(
    context: context,
    showDragHandle: true,
    builder: (sheetContext) {
      return SafeArea(
        child: Padding(
          padding: const EdgeInsets.only(bottom: 8),
          child: ListView(
            shrinkWrap: true,
            children: [
              ListTile(
                title: Text(
                  track.title,
                  maxLines: 1,
                  overflow: TextOverflow.ellipsis,
                ),
                subtitle: Text(
                  '${track.artistName} � ${track.albumTitle}',
                  maxLines: 1,
                  overflow: TextOverflow.ellipsis,
                ),
              ),
              const Divider(height: 1),
              ListTile(
                leading: const Icon(Icons.play_arrow_rounded),
                title: const Text('Play'),
                enabled: playback != null,
                onTap: playback == null
                    ? null
                    : () {
                        Navigator.of(sheetContext).pop();
                        unawaited(playback.playTrack(track));
                      },
              ),
              ListTile(
                leading: const Icon(Icons.playlist_play_rounded),
                title: const Text('Play next'),
                enabled: playback != null,
                onTap: playback == null
                    ? null
                    : () {
                        Navigator.of(sheetContext).pop();
                        unawaited(playback.playNext(track));
                      },
              ),
              ListTile(
                leading: const Icon(Icons.queue_music_rounded),
                title: const Text('Add to queue'),
                enabled: playback != null,
                onTap: playback == null
                    ? null
                    : () {
                        Navigator.of(sheetContext).pop();
                        unawaited(playback.addToQueue(track));
                      },
              ),
              ListTile(
                leading: const Icon(Icons.playlist_add_rounded),
                title: const Text('Add to playlist'),
                onTap: () {
                  Navigator.of(sheetContext).pop();
                  unawaited(showAddToPlaylistFlow(context, track));
                },
              ),
              ListTile(
                leading: const Icon(Icons.album_outlined),
                title: const Text('Go to album'),
                onTap: () {
                  Navigator.of(sheetContext).pop();

                  Navigator.of(context).push(
                    MaterialPageRoute<void>(
                      builder: (_) => AlbumProfilePage(
                        albumId: track.albumId,
                        initialAlbumTitle: track.albumTitle,
                      ),
                    ),
                  );
                },
              ),
              ListTile(
                leading: const Icon(Icons.person_outline_rounded),
                title: const Text('Go to artist'),
                onTap: () {
                  Navigator.of(sheetContext).pop();

                  Navigator.of(context).push(
                    MaterialPageRoute<void>(
                      builder: (_) => ArtistProfilePage(
                        artistId: track.artistId,
                        initialArtistName: track.artistName,
                      ),
                    ),
                  );
                },
              ),
            ],
          ),
        ),
      );
    },
  );
}

String _formatListeningDuration(Duration duration) {
  final totalMinutes = duration.inMinutes;

  if (totalMinutes < 60) {
    return '$totalMinutes min listening';
  }

  final hours = totalMinutes ~/ 60;
  final minutes = totalMinutes % 60;

  if (minutes == 0) {
    return '${hours}h listening';
  }

  return '${hours}h ${minutes}m listening';
}

class HomeScope extends StatefulWidget {
  const HomeScope({super.key, required this.child, this.now});

  final Widget child;
  final HomeControllerNow? now;

  @override
  State<HomeScope> createState() => _HomeScopeState();
}

class _HomeScopeState extends State<HomeScope> {
  LibraryQueryRepository? _library;
  QualifiedPlayRepository? _qualifiedPlays;
  PlaybackSessionRepository? _playbackSessions;
  PlaybackListeningSegmentRepository? _listeningSegments;

  HomeController? _controller;

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();

    final library = context.watch<LibraryQueryRepository?>();
    final qualifiedPlays = context.watch<QualifiedPlayRepository?>();
    final playbackSessions = context.watch<PlaybackSessionRepository?>();
    final listeningSegments = context
        .watch<PlaybackListeningSegmentRepository?>();

    if (identical(library, _library) &&
        identical(qualifiedPlays, _qualifiedPlays) &&
        identical(playbackSessions, _playbackSessions) &&
        identical(listeningSegments, _listeningSegments)) {
      return;
    }

    _library = library;
    _qualifiedPlays = qualifiedPlays;
    _playbackSessions = playbackSessions;
    _listeningSegments = listeningSegments;

    _recreateController();
  }

  @override
  void didUpdateWidget(HomeScope oldWidget) {
    super.didUpdateWidget(oldWidget);

    if (!identical(oldWidget.now, widget.now)) {
      _recreateController();
    }
  }

  void _recreateController() {
    final previousController = _controller;
    _controller = null;

    previousController?.dispose();

    final library = _library;
    final qualifiedPlays = _qualifiedPlays;
    final playbackSessions = _playbackSessions;
    final listeningSegments = _listeningSegments;

    if (library == null ||
        qualifiedPlays == null ||
        playbackSessions == null ||
        listeningSegments == null) {
      return;
    }

    final service = HomeDataService(
      library: library,
      analyticsRankings: AnalyticsEntityRankingService(
        qualifiedPlays: qualifiedPlays,
      ),
      artistListeningDurations: AnalyticsArtistListeningDurationService(
        playbackSessions: playbackSessions,
        listeningSegments: listeningSegments,
      ),
    );

    final controller = HomeController(service: service, now: widget.now);

    _controller = controller;

    unawaited(controller.load());
  }

  @override
  Widget build(BuildContext context) {
    return ChangeNotifierProvider<HomeController?>.value(
      value: _controller,
      child: widget.child,
    );
  }

  @override
  void dispose() {
    _controller?.dispose();
    super.dispose();
  }

String _formatMomentContext(ChartMomentFeedItem item) {
  if (item.isLive) {
    final date = item.occurredAt.toLocal();

    return '${_monthName(date.month)} ${date.day}, ${date.year}';
  }

  return 'Chart week ${_formatMomentWeek(item.weekStartUtc, item.weekEndUtc)}';
}

String _formatMomentWeek(DateTime weekStartUtc, DateTime weekEndUtc) {
  final start = weekStartUtc.toUtc();
  final inclusiveEnd = weekEndUtc.toUtc().subtract(const Duration(days: 1));

  final sameMonth =
      start.year == inclusiveEnd.year && start.month == inclusiveEnd.month;

  if (sameMonth) {
    return '${_monthName(start.month)} '
        '${start.day}\u2013${inclusiveEnd.day}, '
        '${start.year}';
  }

  if (start.year == inclusiveEnd.year) {
    return '${_monthName(start.month)} ${start.day} \u2013 '
        '${_monthName(inclusiveEnd.month)} ${inclusiveEnd.day}, '
        '${start.year}';
  }

  return '${_monthName(start.month)} ${start.day}, ${start.year} \u2013 '
      '${_monthName(inclusiveEnd.month)} '
      '${inclusiveEnd.day}, ${inclusiveEnd.year}';
}

Future<void> _showWeekCalendar(
  BuildContext context,
  ChartsController controller,
) async {
  final history = controller.history;

  if (history.isEmpty) {
    return;
  }

  final availableDates = {
    for (final snapshot in history)
      DateUtils.dateOnly(snapshot.weekStartUtc.toLocal()),
  };

  final earliest = history.last.weekStartUtc.toLocal();
  final latest = history.first.weekStartUtc.toLocal();

  final selected = await showDatePicker(
    context: context,
    helpText: 'Choose a Sonus Hot 100 week',
    initialDate:
        controller.selectedHistoricalSnapshot?.weekStartUtc.toLocal() ?? latest,
    firstDate: DateUtils.dateOnly(earliest),
    lastDate: DateUtils.dateOnly(latest),
    selectableDayPredicate: (date) {
      return availableDates.contains(DateUtils.dateOnly(date));
    },
  );

  if (selected == null) {
    return;
  }

  for (final snapshot in history) {
    final localStart = DateUtils.dateOnly(snapshot.weekStartUtc.toLocal());

    if (localStart == DateUtils.dateOnly(selected)) {
      controller.selectHistoricalWeek(snapshot.weekStartUtc);
      return;
    }
  }
}

String _formatChartWeek(ChartSnapshot snapshot) {
  final start = snapshot.weekStartUtc.toUtc();

  final inclusiveEnd = snapshot.weekEndUtc.toUtc().subtract(
    const Duration(days: 1),
  );

  final sameMonth =
      start.year == inclusiveEnd.year && start.month == inclusiveEnd.month;

  if (sameMonth) {
    return '${_monthName(start.month)} '
        '${start.day}\u2013${inclusiveEnd.day}, '
        '${start.year}';
  }

  if (start.year == inclusiveEnd.year) {
    return '${_monthName(start.month)} ${start.day} \u2013 '
        '${_monthName(inclusiveEnd.month)} ${inclusiveEnd.day}, '
        '${start.year}';
  }

  return '${_monthName(start.month)} ${start.day}, ${start.year} \u2013 '
      '${_monthName(inclusiveEnd.month)} '
      '${inclusiveEnd.day}, ${inclusiveEnd.year}';
}

String _monthName(int month) {
  return const [
    'January',
    'February',
    'March',
    'April',
    'May',
    'June',
    'July',
    'August',
    'September',
    'October',
    'November',
    'December',
  ][month - 1];
}

class _ChartsScopeState extends State<ChartsScope> {
  AppDatabase? _database;
  LibraryQueryRepository? _library;
  QualifiedPlayRepository? _qualifiedPlays;
  PlaybackSessionRepository? _playbackSessions;

  ChartsController? _controller;

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();

    final database = context.watch<AppDatabase>();
    final library = context.watch<LibraryQueryRepository?>();
    final qualifiedPlays = context.watch<QualifiedPlayRepository?>();
    final playbackSessions = context.watch<PlaybackSessionRepository?>();

    if (identical(database, _database) &&
        identical(library, _library) &&
        identical(qualifiedPlays, _qualifiedPlays) &&
        identical(playbackSessions, _playbackSessions)) {
      return;
    }

    _database = database;
    _library = library;
    _qualifiedPlays = qualifiedPlays;
    _playbackSessions = playbackSessions;

    _recreateController();
  }

  @override
  void didUpdateWidget(ChartsScope oldWidget) {
    super.didUpdateWidget(oldWidget);

    if (!identical(oldWidget.now, widget.now)) {
      _recreateController();
    }
  }

  void _recreateController() {
    final previousController = _controller;
    _controller = null;

    previousController?.dispose();

    final database = _database;
    final library = _library;
    final qualifiedPlays = _qualifiedPlays;
    final playbackSessions = _playbackSessions;

    if (database == null ||
        library == null ||
        qualifiedPlays == null ||
        playbackSessions == null) {
      return;
    }

    final charts = SqliteChartRepository(database);

    final moments = SqliteChartMomentRepository(database);
    final builder = ChartSnapshotBuilderService(
      qualifiedPlays: qualifiedPlays,
      library: library,
      charts: charts,
      playbackSessions: playbackSessions,
    );

    final liveCharts = LiveChartService(builder: builder);
    final finalization = ChartFinalizationService(
      builder: builder,
      charts: charts,
      qualifiedPlays: qualifiedPlays,
      moments: moments,
      momentsActivationWeekStartUtc: DateTime(2026, 9, 27).toUtc(),
    );

    final controller = ChartsController(
      liveCharts: liveCharts,
      finalization: finalization,
      charts: charts,
      library: library,
      moments: moments,
      now: widget.now,
    );

    _controller = controller;

    unawaited(controller.load());
  }

  @override
  Widget build(BuildContext context) {
    return ChangeNotifierProvider<ChartsController?>.value(
      value: _controller,
      child: widget.child,
    );
  }

  @override
  void dispose() {
    _controller?.dispose();
    super.dispose();
  }
}

  }

class ListeningHistoryController extends ChangeNotifier {
  ListeningHistoryController({required this._service, int pageSize = 50})
    : _pageSize = _validatedPageSize(pageSize);

  final ListeningHistoryService _service;
  final int _pageSize;

  List<ListeningHistoryEntry> _entries = const [];
  ListeningHistoryCursor? _nextCursor;

  DateTime? _startedAtOrAfter;
  DateTime? _startedBefore;

  bool _isLoading = false;
  bool _isLoadingMore = false;
  bool _hasMore = false;
  Object? _error;

  int _requestGeneration = 0;

  List<ListeningHistoryEntry> get entries => _entries;

  bool get isLoading => _isLoading;

  bool get isLoadingMore => _isLoadingMore;

  bool get hasMore => _hasMore;

  Object? get error => _error;

  bool get hasError => _error != null;

  bool get isEmpty =>
      !_isLoading && !_isLoadingMore && _error == null && _entries.isEmpty;

  DateTime? get startedAtOrAfter => _startedAtOrAfter;

  DateTime? get startedBefore => _startedBefore;

  bool get hasDateFilter => _startedAtOrAfter != null || _startedBefore != null;

  Future<void> load() {
    if (_isLoading) {
      return Future.value();
    }

    return _loadFirstPage();
  }

  Future<void> refresh() {
    return _loadFirstPage();
  }

  Future<void> setDateRange({
    DateTime? startedAtOrAfter,
    DateTime? startedBefore,
  }) async {
    if (startedAtOrAfter != null &&
        startedBefore != null &&
        !startedAtOrAfter.isBefore(startedBefore)) {
      throw ArgumentError(
        'startedAtOrAfter must be earlier than startedBefore.',
      );
    }

    _startedAtOrAfter = startedAtOrAfter;
    _startedBefore = startedBefore;

    await _loadFirstPage();
  }

  Future<void> clearDateRange() {
    return setDateRange();
  }

  Future<void> loadMore() async {
    if (_isLoading || _isLoadingMore || !_hasMore || _nextCursor == null) {
      return;
    }

    final generation = _requestGeneration;
    final cursor = _nextCursor;

    _isLoadingMore = true;
    _error = null;
    notifyListeners();

    try {
      final page = await _service.getPage(
        startedAtOrAfter: _startedAtOrAfter,
        startedBefore: _startedBefore,
        before: cursor,
        limit: _pageSize,
      );

      if (generation != _requestGeneration) {
        return;
      }

      _entries = List<ListeningHistoryEntry>.unmodifiable([
        ..._entries,
        ...page.entries,
      ]);

      _nextCursor = page.nextCursor;
      _hasMore = page.hasMore;
    } catch (error) {
      if (generation != _requestGeneration) {
        return;
      }

      _error = error;
    } finally {
      if (generation == _requestGeneration) {
        _isLoadingMore = false;
        notifyListeners();
      }
    }
  }

  Future<void> _loadFirstPage() async {
    final generation = ++_requestGeneration;

    _isLoading = true;
    _isLoadingMore = false;
    _error = null;

    notifyListeners();

    try {
      final page = await _service.getPage(
        startedAtOrAfter: _startedAtOrAfter,
        startedBefore: _startedBefore,
        limit: _pageSize,
      );

      if (generation != _requestGeneration) {
        return;
      }

      _entries = List<ListeningHistoryEntry>.unmodifiable(page.entries);
      _nextCursor = page.nextCursor;
      _hasMore = page.hasMore;
    } catch (error) {
      if (generation != _requestGeneration) {
        return;
      }

      _error = error;
    } finally {
      if (generation == _requestGeneration) {
        _isLoading = false;
        notifyListeners();
      }
    }
  }

  static int _validatedPageSize(int pageSize) {
    if (pageSize <= 0) {
      throw ArgumentError.value(
        pageSize,
        'pageSize',
        'Must be greater than zero.',
      );
    }

    return pageSize;
  }
}
